const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
function playSound(type) {
  if(audioCtx.state === 'suspended') audioCtx.resume();
  const osc = audioCtx.createOscillator(); const gainNode = audioCtx.createGain();
  osc.connect(gainNode); gainNode.connect(audioCtx.destination);
  if (type === 'correct') {
    osc.type = 'sine'; osc.frequency.setValueAtTime(800, audioCtx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(1200, audioCtx.currentTime + 0.1);
    gainNode.gain.setValueAtTime(0.3, audioCtx.currentTime);
    gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.2);
    osc.start(); osc.stop(audioCtx.currentTime + 0.2);
  } else if (type === 'wrong') {
    osc.type = 'sawtooth'; osc.frequency.setValueAtTime(300, audioCtx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(150, audioCtx.currentTime + 0.2);
    gainNode.gain.setValueAtTime(0.3, audioCtx.currentTime);
    gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);
    osc.start(); osc.stop(audioCtx.currentTime + 0.3);
  }
}

function showModal(message, type = 'alert', callback = null, titleKey = 'Makluman') {
  document.getElementById('modalTitle').textContent = typeof titleKey === 'string' && titleKey !== 'Makluman' ? titleKey : (state.lang === 'en' ? 'Notice' : 'Makluman');
  document.getElementById('modalMessage').innerHTML = message;
  document.getElementById('customModal').classList.add('active');
  const btnCancel = document.getElementById('modalCancel');
  const btnConfirm = document.getElementById('modalConfirm');
  btnConfirm.onclick = () => { closeModal(); if (callback) callback(); };
  if (type === 'confirm') { btnCancel.style.display = 'block'; btnCancel.onclick = closeModal; } 
  else { btnCancel.style.display = 'none'; }
}
function closeModal() { document.getElementById('customModal').classList.remove('active'); }

const STATE_KEY = 'scaffsafe_pro_state';
let state = {
  lang: 'ms', currentLevel: 1, currentModuleId: null, currentQuizIndex: 0, currentQuizQuestions: [],
  quizAnswers: [], quizMode: 'module', quizStartTime: 0, selectedOption: -1,
  stats: { completedModules: [], quizzesTaken: 0, totalCorrect: 0, totalQuestions: 0, streak: 1, lastActiveDate: new Date().toDateString() }
};

function saveState() { try { localStorage.setItem(STATE_KEY, JSON.stringify(state)); } catch(e) { } }
function loadState() {
  try {
    const saved = localStorage.getItem(STATE_KEY);
    if (saved) { state = { ...state, ...JSON.parse(saved) }; updateStreak(); }
  } catch(e) { }
}

function toggleLang() {
  state.lang = state.lang === 'ms' ? 'en' : 'ms';
  saveState();
  applyLanguage();
}

function applyLanguage() {
  document.getElementById('btn-lang-ms').className = state.lang === 'ms' ? 'active' : '';
  document.getElementById('btn-lang-en').className = state.lang === 'en' ? 'active' : '';
  
  document.querySelectorAll('[data-i18n]').forEach(el => {
    const key = el.getAttribute('data-i18n');
    if (UI_TEXT[key] && UI_TEXT[key][state.lang]) {
      if (el.tagName === 'INPUT' && el.type === 'text') {
        el.placeholder = UI_TEXT[key][state.lang];
      } else {
        el.innerHTML = UI_TEXT[key][state.lang];
      }
    }
  });

  updateDashboard();
  randomizeFact();
  
  if (document.getElementById('screen-levels').classList.contains('active')) showLevelModules(state.currentLevel);
  if (document.getElementById('screen-lesson').classList.contains('active')) openModule(state.currentModuleId);
  if (document.getElementById('screen-quiz').classList.contains('active')) loadQuestion();
  if (document.getElementById('screen-checklist').classList.contains('active')) initChecklist();
  if (document.getElementById('screen-progress').classList.contains('active')) updateProgressScreen();
  if (document.getElementById('screen-calculator').classList.contains('active')) calculateLoad();
  
  document.documentElement.lang = state.lang;
}

function updateStreak() {
  const today = new Date().toDateString();
  const lastDate = state.stats.lastActiveDate;
  if (lastDate !== today) {
    const yesterday = new Date(Date.now() - 86400000).toDateString();
    if (lastDate === yesterday) state.stats.streak += 1; else state.stats.streak = 1;
    state.stats.lastActiveDate = today; saveState();
  }
}

function startModuleQuiz() {
  const mod = MODULES[state.currentLevel].find(m => m.id === state.currentModuleId);
  if (!mod) return;
  state.quizMode = 'module'; state.currentQuizQuestions = [...mod.quiz];
  state.currentQuizIndex = 0; state.quizAnswers = []; state.quizStartTime = Date.now(); state.selectedOption = -1;
  showScreen('quiz'); loadQuestion(); startTimer(false);
}

function startDailyQuiz() {
  const allQuestions = [];
  MODULES[1].forEach(mod => { mod.quiz.forEach(q => allQuestions.push(q)); });
  const shuffled = allQuestions.sort(() => Math.random() - 0.5);
  state.quizMode = 'daily'; state.currentQuizQuestions = shuffled.slice(0, 5);
  state.currentQuizIndex = 0; state.quizAnswers = []; state.quizStartTime = Date.now();
  showScreen('quiz'); loadQuestion(); startTimer(false);
}

function startMockExam() {
  const allQuestions = [];
  MODULES[1].forEach(mod => { mod.quiz.forEach(q => allQuestions.push(q)); });
  if (allQuestions.length < 10) { showModal(state.lang === 'ms' ? 'Sistem sedang kemaskini data bank soalan.' : 'System is updating question bank data.'); return; }
  const shuffled = allQuestions.sort(() => Math.random() - 0.5);
  state.quizMode = 'mock'; state.currentQuizQuestions = shuffled.slice(0, 10); 
  state.currentQuizIndex = 0; state.quizAnswers = []; state.quizStartTime = Date.now();
  showScreen('quiz'); loadQuestion(); startTimer(true);
}

let timerInterval = null;
function startTimer(isCountdown) {
  clearInterval(timerInterval);
  const duration = isCountdown ? (20 * 60) : 0; 
  const startTime = Date.now();
  
  timerInterval = setInterval(() => {
    const elapsed = Math.floor((Date.now() - startTime) / 1000);
    let display;
    if (isCountdown) {
      const remaining = duration - elapsed;
      if (remaining <= 0) {
        clearInterval(timerInterval); playSound('wrong');
        showModal(state.lang === 'ms' ? "Masa Tamat!" : "Time's Up!", "alert", finishQuiz); return;
      }
      const m = Math.floor(remaining / 60); const s = remaining % 60;
      display = String(m).padStart(2,'0') + ':' + String(s).padStart(2,'0');
      if(remaining < 180) document.getElementById('quizTimer').style.color = "var(--danger)";
    } else {
      const m = Math.floor(elapsed / 60); const s = elapsed % 60;
      display = String(m).padStart(2,'0') + ':' + String(s).padStart(2,'0');
      document.getElementById('quizTimer').style.color = "var(--accent)"; 
    }
    document.getElementById('quizTimer').textContent = display;
  }, 1000);
}

function loadQuestion() {
  if (state.currentQuizQuestions.length === 0) return;
  const q = state.currentQuizQuestions[state.currentQuizIndex];
  document.getElementById('currentQNum').textContent = state.currentQuizIndex + 1;
  document.getElementById('totalQNum').textContent = state.currentQuizQuestions.length;
  document.getElementById('quizQuestion').textContent = q.q[state.lang];
  document.getElementById('explanation').classList.remove('show');
  
  const optionsDiv = document.getElementById('quizOptions');
  optionsDiv.innerHTML = ''; state.selectedOption = -1;
  const letters = ['A', 'B', 'C', 'D'];
  q.options[state.lang].forEach((opt, i) => {
    const btn = document.createElement('button');
    btn.className = 'quiz-option'; btn.onclick = () => selectOption(i);
    btn.innerHTML = `<span class="option-letter">${letters[i]}</span><span>${opt}</span>`;
    optionsDiv.appendChild(btn);
  });
  document.getElementById('nextBtn').disabled = true;
  document.getElementById('nextBtn').textContent = state.lang === 'ms' ? 'Semak Jawapan' : 'Check Answer';
}

function selectOption(idx) {
  if(audioCtx.state === 'suspended') audioCtx.resume();
  state.selectedOption = idx;
  document.querySelectorAll('.quiz-option').forEach((opt, i) => { opt.classList.toggle('selected', i === idx); });
  document.getElementById('nextBtn').disabled = false;
  document.getElementById('nextBtn').textContent = state.lang === 'ms' ? 'Semak Jawapan' : 'Check Answer';
}

function nextQuestion() {
  const btn = document.getElementById('nextBtn');
  const q = state.currentQuizQuestions[state.currentQuizIndex];
  const isLast = state.currentQuizIndex === state.currentQuizQuestions.length - 1;
  
  if (btn.textContent.includes('Semak') || btn.textContent.includes('Check')) {
    const isCorrect = state.selectedOption === q.answer;
    if(isCorrect) playSound('correct'); else playSound('wrong');

    state.quizAnswers.push({ question: q, selected: state.selectedOption, correct: isCorrect });
    document.querySelectorAll('.quiz-option').forEach((opt, i) => {
      opt.classList.add('disabled'); opt.onclick = null;
      if (i === q.answer) opt.classList.add('correct');
      else if (i === state.selectedOption) opt.classList.add('wrong');
    });
    document.getElementById('explanationText').textContent = q.explanation[state.lang];
    document.getElementById('explanation').classList.add('show');
    btn.textContent = isLast ? (state.lang === 'ms' ? 'Lihat Keputusan' : 'View Results') : (state.lang === 'ms' ? 'Soalan Seterusnya →' : 'Next Question →');
    return;
  }
  
  state.currentQuizIndex++;
  if (state.currentQuizIndex >= state.currentQuizQuestions.length) finishQuiz();
  else loadQuestion();
}

function finishQuiz() {
  clearInterval(timerInterval);
  const correct = state.quizAnswers.filter(a => a.correct).length;
  const total = state.quizAnswers.length;
  const percent = total > 0 ? Math.round((correct / total) * 100) : 0;
  const timeTaken = Math.floor((Date.now() - state.quizStartTime) / 1000);
  
  state.stats.quizzesTaken++; state.stats.totalCorrect += correct; state.stats.totalQuestions += total;
  const passMark = state.quizMode === 'mock' ? 80 : 60; 
  if (state.quizMode === 'module' && percent >= passMark) {
    if (!state.stats.completedModules.includes(state.currentModuleId)) state.stats.completedModules.push(state.currentModuleId);
  }
  saveState();
  
  document.getElementById('resultScore').textContent = percent + '%';
  document.getElementById('statCorrect').textContent = correct;
  document.getElementById('statWrong').textContent = total - correct;
  const m = Math.floor(timeTaken / 60); const s = timeTaken % 60;
  document.getElementById('statTime').textContent = m + ':' + String(s).padStart(2,'0');
  
  const statusEl = document.getElementById('resultStatus');
  if (percent >= passMark) {
    statusEl.innerHTML = state.lang === 'ms' ? `LULUS ✓<br><small style="font-size:0.65rem; opacity:0.8;">MELEPASI TAHAP MINIMUM (${passMark}%)</small>` : `PASSED ✓<br><small style="font-size:0.65rem; opacity:0.8;">PASSED MINIMUM REQUIREMENT (${passMark}%)</small>`;
    statusEl.className = 'result-status pass'; playSound('correct');
  } else {
    statusEl.innerHTML = state.lang === 'ms' ? `GAGAL ✗<br><small style="font-size:0.65rem; opacity:0.8;">SKOR MINIMUM: ${passMark}%</small>` : `FAILED ✗<br><small style="font-size:0.65rem; opacity:0.8;">MINIMUM SCORE: ${passMark}%</small>`;
    statusEl.className = 'result-status fail'; playSound('wrong');
  }
  showScreen('result');
}

function exitQuiz() {
  showModal(state.lang === 'ms' ? 'Amaran: Rekod sesi ini akan terpadam. Teruskan?' : 'Warning: Records for this session will be lost. Continue?', 'confirm', function() {
    clearInterval(timerInterval); goHome();
  });
}
function retryQuiz() { if (state.quizMode === 'module') startModuleQuiz(); else if (state.quizMode === 'daily') startDailyQuiz(); else startMockExam(); }

  const plank = (parseFloat(document.getElementById('calcPlank').value) || 0) * 5.4;
  const guardrail = (parseFloat(document.getElementById('calcGuardrail').value) || 0) * 2.55;
  const brace = (parseFloat(document.getElementById('calcBrace').value) || 0) * 5.9;
  const base = (parseFloat(document.getElementById('calcBase').value) || 0) * 1.4;
  const ladder = (parseFloat(document.getElementById('calcLadder').value) || 0) * 2.9;
  const lain = parseFloat(document.getElementById('calcLain').value) || 0;
  
  const deadLoad = plank + guardrail + brace + base + ladder + lain;
  const dutyLoad = parseFloat(document.getElementById('calcDuty').value) || 0;
  const totalLoad = deadLoad + dutyLoad;
  
  document.getElementById('calcResult').innerHTML = totalLoad.toFixed(1) + ' <span style="font-size:1.2rem; color:var(--text-dim);">kg</span>';
  
  const boxEl = document.getElementById('calcResultBox'); const warningEl = document.getElementById('calcWarning');
  if (totalLoad > 2000) {
    boxEl.style.borderColor = "var(--danger)"; boxEl.style.boxShadow = "0 10px 30px rgba(255,61,0,0.3)";
    warningEl.style.display = "block"; warningEl.style.color = "var(--danger)"; warningEl.style.backgroundColor = "rgba(255,61,0,0.1)";
    warningEl.textContent = state.lang === 'ms' ? "AMARAN: Beban melebihi 2000kg. Risiko Base Jack." : "WARNING: Load exceeds 2000kg. Base Jack risk.";
  } else if (totalLoad > 1000) {
    boxEl.style.borderColor = "var(--warning)"; boxEl.style.boxShadow = "0 10px 30px rgba(255,196,0,0.2)";
    warningEl.style.display = "block"; warningEl.style.color = "var(--warning)"; warningEl.style.backgroundColor = "rgba(255,196,0,0.1)";
    warningEl.textContent = state.lang === 'ms' ? "Nota: Beban Sederhana Tinggi." : "Note: Medium-High Load.";
  } else {
    boxEl.style.borderColor = "var(--accent)"; boxEl.style.boxShadow = "0 10px 30px var(--surface-glow)"; warningEl.style.display = "none";
  }
}

function initChecklist() {
  const container = document.getElementById('checklistItemsContainer'); container.innerHTML = '';
  let savedChecks = []; try { savedChecks = JSON.parse(localStorage.getItem('scaffsafe_checklist') || '[]'); } catch(e){}

  checklistDataContent[state.lang].forEach((text, i) => {
    const isChecked = savedChecks[i] ? 'checked' : '';
    const label = document.createElement('label'); label.className = 'interactive-check';
    label.innerHTML = `<input type="checkbox" id="chk-${i}" onchange="saveChecklist()" ${isChecked}> <span>${text}</span>`;
    container.appendChild(label);
  });
  updateChecklistDate();
}

function saveChecklist() {
  const checks = [];
  checklistDataContent[state.lang].forEach((_, i) => { checks[i] = document.getElementById(`chk-${i}`).checked; });
  localStorage.setItem('scaffsafe_checklist', JSON.stringify(checks));
}

function resetChecklist() {
  showModal(state.lang === 'ms' ? 'Padam rekod checklist harian ini?' : 'Delete today\'s checklist record?', 'confirm', () => {
    localStorage.removeItem('scaffsafe_checklist'); initChecklist();
  });
}

function updateChecklistDate() {
  const dt = new Date();
  document.getElementById('checklistDateDisplay').textContent = (state.lang === 'ms' ? "Tarikh Inspeksi: " : "Inspection Date: ") + dt.toLocaleDateString() + " " + dt.toLocaleTimeString();
}

function printChecklist() { updateChecklistDate(); document.body.classList.add('print-checklist-mode'); window.print(); document.body.classList.remove('print-checklist-mode'); }

function doSearch() {
  const query = document.getElementById('searchInput').value.toLowerCase().trim();
  const results = document.getElementById('searchResults'); results.innerHTML = '';
  if (!query || query.length < 2) {
    results.innerHTML = `<div style="text-align:center; padding:2rem 0; color:var(--text-dim); font-size:0.9rem;"><svg class="svg-icon" style="width:40px;height:40px; opacity:0.5; margin-bottom:1rem;" viewBox="0 0 24 24"><circle cx="11" cy="11" r="8"></circle><line x1="21" y1="21" x2="16.65" y2="16.65"></line></svg><br>${state.lang==='ms' ? 'Gunakan ruangan di atas untuk mencari maklumat pantas.' : 'Use the field above to quickly search for info.'}</div>`;
    return;
  }
  
  const found = [];
  MODULES[1].forEach((mod, idx) => {
    if (mod.title[state.lang].toLowerCase().includes(query) || mod.content[state.lang].toLowerCase().includes(query)) found.push({ mod, idx });
  });
  
  if (found.length === 0) {
    results.innerHTML = `<div style="padding:1rem; color:var(--warning); text-align:center; font-family: JetBrains Mono, monospace; font-size:0.85rem; background: rgba(255,196,0,0.1); border-radius: 8px;">${state.lang==='ms'?'Tiada padanan rujukan ditemui.':'No matches found.'}</div>`;
    return;
  }
  
  found.forEach(({ mod, idx }) => {
    const btn = document.createElement('button'); btn.className = 'module-item'; btn.onclick = () => openModule(mod.id);
    btn.innerHTML = `<span class="module-number">Ref ${String(idx + 1).padStart(2, '0')}</span><span class="module-title">${mod.title[state.lang]}</span><span class="module-status" style="color:var(--accent);">Baca →</span>`;
    results.appendChild(btn);
  });
}
